cmake_minimum_required(VERSION 3.18)
project(brotli_metadata_reference LANGUAGES C CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(PORT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(BROTLI "${PORT_ROOT}/upstream/libjxl/third_party/brotli")
option(ORACLE_SANITIZERS "Instrument the entire original Brotli graph" OFF)
if(ORACLE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BROTLI_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BROTLI_DISABLE_TESTS ON CACHE BOOL "" FORCE)
add_subdirectory("${BROTLI}" brotli EXCLUDE_FROM_ALL)
add_executable(brotli_metadata_oracle ../brotli_metadata.cc)
target_compile_features(brotli_metadata_oracle PRIVATE cxx_std_17)
target_include_directories(brotli_metadata_oracle PRIVATE "${BROTLI}/c/include" "${BROTLI}/c")
target_link_libraries(brotli_metadata_oracle PRIVATE brotlidec brotlienc)
if(APPLE)
  target_link_options(brotli_metadata_oracle PRIVATE "-Wl,-map,${CMAKE_CURRENT_BINARY_DIR}/brotli_metadata.map")
else()
  target_link_options(brotli_metadata_oracle PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/brotli_metadata.map")
endif()
