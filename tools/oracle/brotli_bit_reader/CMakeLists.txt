cmake_minimum_required(VERSION 3.18)
project(brotli_bit_reader_observer LANGUAGES C)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(PORT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(BROTLI_ROOT "${PORT_ROOT}/upstream/libjxl/third_party/brotli")
option(ORACLE_SANITIZERS "Instrument every original source and observer" OFF)
if(ORACLE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BROTLI_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BROTLI_DISABLE_TESTS ON CACHE BOOL "" FORCE)
add_subdirectory("${BROTLI_ROOT}" brotli EXCLUDE_FROM_ALL)
add_executable(brotli_bit_reader_observer ../brotli_bit_reader.c)
target_include_directories(brotli_bit_reader_observer PRIVATE "${BROTLI_ROOT}")
target_link_libraries(brotli_bit_reader_observer PRIVATE brotlidec)
