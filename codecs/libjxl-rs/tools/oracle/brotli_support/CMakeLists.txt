cmake_minimum_required(VERSION 3.18)
project(brotli_support_reference LANGUAGES C CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(PORT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(BROTLI "${PORT_ROOT}/upstream/libjxl/third_party/brotli")
option(ORACLE_SANITIZERS "Instrument the entire original Brotli graph" OFF)
if(ORACLE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BROTLI_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BROTLI_DISABLE_TESTS ON CACHE BOOL "" FORCE)
add_subdirectory("${BROTLI}" brotli EXCLUDE_FROM_ALL)

# Join the independently observed adapters without changing their bodies or
# the original archives. Each executable keeps its own include-TU/test bridge;
# no observer definitions affect another adapter or an archive compilation.
function(observer target)
  add_executable(${target} ${ARGN})
  target_compile_features(${target} PRIVATE c_std_11 cxx_std_17)
  target_include_directories(${target} PRIVATE "${BROTLI}" "${BROTLI}/c" "${BROTLI}/c/include")
  target_link_libraries(${target} PRIVATE brotlidec brotlienc brotlicommon)
  if(APPLE)
    target_link_options(${target} PRIVATE "-Wl,-map,${CMAKE_CURRENT_BINARY_DIR}/${target}.map")
  else()
    target_link_options(${target} PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/${target}.map")
  endif()
endfunction()
