cmake_minimum_required(VERSION 3.18)
project(brotli_stream_blocks_reference LANGUAGES C)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(PORT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(BROTLI "${PORT_ROOT}/upstream/libjxl/third_party/brotli")
option(ORACLE_SANITIZERS "Instrument the entire original Brotli graph" OFF)
if(ORACLE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BROTLI_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BROTLI_DISABLE_TESTS ON CACHE BOOL "" FORCE)
add_subdirectory("${BROTLI}" brotli EXCLUDE_FROM_ALL)
add_executable(brotli_stream_blocks_oracle ../brotli_stream_blocks.c)
target_include_directories(brotli_stream_blocks_oracle PRIVATE "${BROTLI}" "${BROTLI}/c/include")
target_link_libraries(brotli_stream_blocks_oracle PRIVATE brotlidec)
if(APPLE)
  target_link_options(brotli_stream_blocks_oracle PRIVATE "-Wl,-map,${CMAKE_CURRENT_BINARY_DIR}/brotli_stream_blocks.map")
else()
  target_link_options(brotli_stream_blocks_oracle PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/brotli_stream_blocks.map")
endif()
