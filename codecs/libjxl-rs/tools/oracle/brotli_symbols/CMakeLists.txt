cmake_minimum_required(VERSION 3.18)
project(brotli_symbols_reference LANGUAGES C)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(PORT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(BROTLI "${PORT_ROOT}/upstream/libjxl/third_party/brotli")
option(ORACLE_SANITIZERS "Instrument the entire original Brotli graph" OFF)
if(ORACLE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BROTLI_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BROTLI_DISABLE_TESTS ON CACHE BOOL "" FORCE)
add_subdirectory("${BROTLI}" brotli EXCLUDE_FROM_ALL)
foreach(target brotli_symbols_oracle brotli_symbols_default)
  add_executable(${target} ../brotli_symbols.c)
  target_compile_features(${target} PRIVATE c_std_11)
  target_include_directories(${target} PRIVATE "${BROTLI}/c/include" "${BROTLI}/c")
  target_link_libraries(${target} PRIVATE brotlidec brotlicommon)
  if(APPLE)
    target_link_options(${target} PRIVATE "-Wl,-map,${CMAKE_CURRENT_BINARY_DIR}/${target}.map")
  else()
    target_link_options(${target} PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/${target}.map")
  endif()
endforeach()
# Only the dedicated include-TU observer gains the two existing test bridges.
# The archive and the independent complete-stream target keep default flags.
target_compile_definitions(brotli_symbols_oracle PRIVATE BROTLI_TEST=1)
target_compile_definitions(brotli_symbols_default PRIVATE SYMBOLS_DEFAULT=1)
